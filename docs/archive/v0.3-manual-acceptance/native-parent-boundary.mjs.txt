import { spawn } from 'node:child_process';
import { mkdirSync,writeFileSync,readFileSync,existsSync } from 'node:fs';
import path from 'node:path';
const cwd='/workspaces/development/fuzzing-acceptance',directory=path.join(cwd,'.capri/core/acceptance'),marker=path.join(directory,'boundary-'+Date.now()+'.jsonl');mkdirSync(directory,{recursive:true});
const prompt=`Perform a bounded native parent-turn completion test. Launch exactly one native fuzzing child with its installed role. No proxy, browser, fuzzing or external target access. Child task: use a shell command to append JSON with stage=1 and Date.now() to ${marker}, then wait 20 seconds using a tool, append JSON stage=2 with Date.now() to the same file, and finish. This is a synthetic lifecycle test; direct shell is permitted for the marker only. Main task: poll that file until stage 1 exists, then immediately send your final answer WITHOUT waiting for the child. Do not close or interrupt it. We will inspect whether stage 2 appears after your final turn. Do not start any other specialist. If native launch fails, report it and stop.`;
const child=spawn('codex',['exec','--json','--skip-git-repo-check','-C',cwd,prompt],{stdio:['ignore','pipe','pipe']});let output='',errors='';child.stdout.on('data',b=>output+=b);child.stderr.on('data',b=>errors=(errors+b).slice(-4000));
const timer=setTimeout(()=>child.kill('SIGTERM'),180000);const code=await new Promise(r=>child.on('close',r));clearTimeout(timer);const parentExit=Date.now();await new Promise(r=>setTimeout(r,30000));
const markers=existsSync(marker)?readFileSync(marker,'utf8').trim().split('\n').flatMap(s=>{try{return[JSON.parse(s)];}catch{return[];}}):[];
const receipt={time:new Date().toISOString(),exit_code:code,parent_exit:parentExit,markers,continued_after_exec_exit:markers.some(m=>m.stage===2&&m.time>parentExit),boundary:'Persisted codex exec process ending; desktop keeps its own app-server alive and requires separate boundary qualification'};
writeFileSync(path.join(directory,'parent-boundary-events.jsonl'),output);writeFileSync(path.join(directory,'parent-boundary-diagnostics.txt'),errors);writeFileSync(path.join(directory,'parent-boundary.json'),JSON.stringify(receipt,null,2));console.log(JSON.stringify(receipt));
